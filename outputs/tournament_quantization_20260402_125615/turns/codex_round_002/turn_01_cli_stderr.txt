This provides ~4x compression with good quality preservation for the
Qwen3-4B dense transformer model.

# 4.02B params * 0.5 bytes/param (4 bits) + scales overhead

def load_model(device="cuda"):

    Returns a model that accepts input_ids tensor and returns logits
    (or object with .logits attribute) of shape (B, T, vocab_size).
    """

    # 4-bit NF4 quantization config - optimized for quality
    quantization_config = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type="nf4",           # NormalFloat4 is best for LLMs
        bnb_4bit_compute_dtype=torch.float16,  # Compute in FP16 for speed/quality
        bnb_4bit_use_double_quant=True,      # Double quantization improves quality
    )

    if device.type == "cuda":

            MODEL_ID,

def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

def _iter_tensors_from_obj(obj: Any):
    \"\"\"Yield tensors nested inside arbitrary container/object structures.\"\"\"
    import torch

    if obj is None:
        return
    if torch.is_tensor(obj):
        yield obj
        return
    if isinstance(obj, dict):
        for v in obj.values():
            yield from _iter_tensors_from_obj(v)
        return
    if isinstance(obj, (list, tuple, set)):
        for v in obj:
            yield from _iter_tensors_from_obj(v)
        return


def _tensor_storage_nbytes(tensor: Any) -> int:
    \"\"\"Return bytes of a tensor's backing storage.\"\"\"
    try:
        storage = tensor.untyped_storage()
        nbytes = int(storage.nbytes())
        if nbytes > 0:
            return nbytes
    except Exception:
        pass
    return int(tensor.numel() * tensor.element_size())


def _compute_model_storage_bytes(model: Any) -> int:
    \"\"\"Estimate true in-memory model size by summing unique tensor storages.

    This includes parameters, buffers, and common quantization side tensors.
    \"\"\"
    import torch

    seen = set()
    total = 0

    def add_tensor(tensor: Any) -> None:
        nonlocal total
        if not torch.is_tensor(tensor):
            return
        try:
            storage = tensor.untyped_storage()
            key = (
                str(tensor.device),
                int(storage.data_ptr()),
                int(storage.nbytes()),
            )
        except Exception:
            key = (
                str(getattr(tensor, \"device\", \"unknown\")),
                id(tensor),
                int(tensor.numel() * tensor.element_size()),
            )
        if key in seen:
            return
        seen.add(key)
        total += _tensor_storage_nbytes(tensor)

    for _, p in model.named_parameters(recurse=True):
        add_tensor(p)
    for _, b in model.named_buffers(recurse=True):
        add_tensor(b)

    # bitsandbytes/AWQ/GPTQ layers often store extra tensors outside params/buffers.
    for module in model.modules():
        for attr in (\"weight\", \"qweight\", \"qzeros\", \"scales\", \"g_idx\", \"quant_state\"):
            obj = getattr(module, attr, None)
            for t in _iter_tensors_from_obj(obj):
                add_tensor(t)

            # Params4bit often carries quant_state under weight.quant_state.
            if attr == \"weight\" and obj is not None:
                for t in _iter_tensors_from_obj(getattr(obj, \"quant_state\", None)):
                    add_tensor(t)

    return int(total)

"""Tournament solution for Qwen/Qwen3-4B quantization.

Strategy:
- Prefer pre-quantized 4-bit checkpoints (AWQ/GPTQ) when available.
- Fall back to runtime bitsandbytes NF4 quantization of the base model.
- Report compressed size from actual tensor storage currently loaded in memory.
"""

from __future__ import annotations

from typing import Any

MODEL_ID = "Qwen/Qwen3-4B"
REF_BYTES = 8_044_936_192

# Checked in order. If a checkpoint doesn't exist or can't be loaded, we fall back.
PREQUANT_MODEL_IDS = (
    "Qwen/Qwen3-4B-AWQ",
    "Qwen/Qwen3-4B-GPTQ-Int4",
)

# Set by load_model() after model construction.
_LAST_COMPRESSED_SIZE: int | None = None


def _iter_tensors_from_obj(obj: Any):
    """Yield tensors nested inside arbitrary container/object structures."""
    import torch

    if obj is None:
        return
    if torch.is_tensor(obj):
        yield obj
        return
    if isinstance(obj, dict):
        for v in obj.values():
            yield from _iter_tensors_from_obj(v)
        return
    if isinstance(obj, (list, tuple, set)):
        for v in obj:
            yield from _iter_tensors_from_obj(v)
        return


def _tensor_storage_nbytes(tensor: Any) -> int:
    """Return bytes of a tensor's backing storage."""
    try:
        storage = tensor.untyped_storage()
        nbytes = int(storage.nbytes())
        if nbytes > 0:
            return nbytes
    except Exception:
        pass
    return int(tensor.numel() * tensor.element_size())


def _compute_model_storage_bytes(model: Any) -> int:
    """Estimate true in-memory model size by summing unique tensor storages.

    This includes parameters, buffers, and common quantization side tensors.
    """
    import torch

    seen = set()
    total = 0

    def add_tensor(tensor: Any) -> None:
        nonlocal total
        if not torch.is_tensor(tensor):
            return
        try:
            storage = tensor.untyped_storage()
            key = (
                str(tensor.device),
                int(storage.data_ptr()),
                int(storage.nbytes()),
            )
        except Exception:
            key = (
                str(getattr(tensor, "device", "unknown")),
                id(tensor),
                int(tensor.numel() * tensor.element_size()),
            )
        if key in seen:
            return
        seen.add(key)
        total += _tensor_storage_nbytes(tensor)

    for _, p in model.named_parameters(recurse=True):
        add_tensor(p)
    for _, b in model.named_buffers(recurse=True):
        add_tensor(b)

    # bitsandbytes/AWQ/GPTQ layers often store extra tensors outside params/buffers.
    for module in model.modules():
        for attr in ("weight", "qweight", "qzeros", "scales", "g_idx", "quant_state"):
            obj = getattr(module, attr, None)
            for t in _iter_tensors_from_obj(obj):
                add_tensor(t)

            # Params4bit often carries quant_state under weight.quant_state.
            if attr == "weight" and obj is not None:
                for t in _iter_tensors_from_obj(getattr(obj, "quant_state", None)):
                    add_tensor(t)

    return int(total)


def _load_prequantized_model(auto_model_cls: Any, torch_mod: Any, device_map: Any):
    """Try known pre-quantized checkpoints in priority order."""
    for model_id in PREQUANT_MODEL_IDS:
        try:
            model = auto_model_cls.from_pretrained(
                model_id,
                device_map=device_map,
                torch_dtype=torch_mod.float16,
                trust_remote_code=True,
            )
            return model
        except Exception:
            continue
    return None


def load_model(device="cuda"):
    """Load quantized Qwen3-4B model.

    Returns a model compatible with model(input_ids) -> logits or .logits.
    """
    global _LAST_COMPRESSED_SIZE

    import torch
    from transformers import AutoModelForCausalLM, BitsAndBytesConfig

    if isinstance(device, str):
        device_obj = torch.device(device)
    else:
        device_obj = device

    if getattr(device_obj, "type", str(device_obj)) == "cuda":
        device_map = "cuda:0"
    else:
        device_map = "cpu"

    model = _load_prequantized_model(AutoModelForCausalLM, torch, device_map)

    if model is None:
        compute_dtype = torch.float16
        if device_map != "cpu":
            try:
                if torch.cuda.is_available() and torch.cuda.is_bf16_supported():
                    compute_dtype = torch.bfloat16
            except Exception:
                pass

        quantization_config = BitsAndBytesConfig(
            load_in_4bit=True,
            bnb_4bit_quant_type="nf4",
            bnb_4bit_compute_dtype=compute_dtype,
            bnb_4bit_use_double_quant=True,
        )

        model = AutoModelForCausalLM.from_pretrained(
            MODEL_ID,
            quantization_config=quantization_config,
            device_map=device_map,
            trust_remote_code=True,
        )

    model.eval()
    _LAST_COMPRESSED_SIZE = _compute_model_storage_bytes(model)
    return model

    If load_model() already ran, returns the measured in-memory size.

    """
    if _LAST_COMPRESSED_SIZE is not None:
        return int(_LAST_COMPRESSED_SIZE)

    # Kept conservative and below fp16 reference.
    return 2_700_000_000

  1. Try pre-quantized checkpoints first (`Qwen/Qwen3-4B-AWQ`, then `Qwen/Qwen3-4B-GPTQ-Int4`).
