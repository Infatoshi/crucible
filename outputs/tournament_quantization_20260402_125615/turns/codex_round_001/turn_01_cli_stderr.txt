def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

MODEL_ID = \"Qwen/Qwen3-4B\"
ARTIFACT_DIR = Path(__file__).resolve().parent / \"quantized_qwen3_4b\"

BnbModel = Optional[nn.Module]


class Int8Linear(nn.Module):
    \"\"\"Simple per-output-channel int8 linear layer.

    This is used only as a fallback when bitsandbytes is unavailable.
    \"\"\"

    def __init__(self, in_features: int, out_features: int, has_bias: bool):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.register_buffer(\"qweight\", torch.zeros(out_features, in_features, dtype=torch.int8))
        self.register_buffer(\"scale\", torch.ones(out_features, dtype=torch.float16))
        if has_bias:
            self.register_buffer(\"bias\", torch.zeros(out_features, dtype=torch.float16))
        else:
            self.bias = None

    @classmethod
    def from_linear(cls, linear: nn.Linear) -> \"Int8Linear\":
        out_features, in_features = linear.weight.shape
        layer = cls(in_features=in_features, out_features=out_features, has_bias=linear.bias is not None)
        with torch.no_grad():
            w = linear.weight.detach().to(torch.float32)
            max_abs = w.abs().amax(dim=1)
            scale = torch.clamp(max_abs / 127.0, min=1e-8)
            q = torch.clamp(torch.round(w / scale.unsqueeze(1)), -127, 127).to(torch.int8)
            layer.qweight.copy_(q)
            layer.scale.copy_(scale.to(torch.float16))
            if linear.bias is not None:
                layer.bias.copy_(linear.bias.detach().to(torch.float16))
        return layer

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        w = self.qweight.to(torch.float32) * self.scale.unsqueeze(1).to(torch.float32)
        bias = self.bias.to(torch.float32) if self.bias is not None else None
        out = F.linear(x.to(torch.float32), w, bias)
        return out.to(x.dtype)


def _to_device_map(device: Union[str, torch.device]):
    if isinstance(device, torch.device):
        device_str = str(device)
    else:
        device_str = str(device)

    if device_str.startswith(\"cuda\"):
        return {\"\": device_str}
    return {\"\": \"cpu\"}


def _dir_size_bytes(path: Path) -> int:
    total = 0
    for root, _, files in os.walk(path):
        for name in files:
            total += (Path(root) / name).stat().st_size
    return total


def _swap_linears_to_int8(module: nn.Module):
    for name, child in list(module.named_children()):
        if isinstance(child, nn.Linear):
            setattr(module, name, Int8Linear.from_linear(child))
        else:
            _swap_linears_to_int8(child)

class Int8Linear(nn.Module):
    """Simple per-output-channel int8 linear layer.

    This is used only as a fallback when bitsandbytes is unavailable.
    """

    def __init__(self, in_features: int, out_features: int, has_bias: bool):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.register_buffer("qweight", torch.zeros(out_features, in_features, dtype=torch.int8))
        self.register_buffer("scale", torch.ones(out_features, dtype=torch.float16))
        if has_bias:
            self.register_buffer("bias", torch.zeros(out_features, dtype=torch.float16))
        else:
            self.bias = None

    @classmethod
    def from_linear(cls, linear: nn.Linear) -> "Int8Linear":
        out_features, in_features = linear.weight.shape
        layer = cls(in_features=in_features, out_features=out_features, has_bias=linear.bias is not None)
        with torch.no_grad():
            w = linear.weight.detach().to(torch.float32)
            max_abs = w.abs().amax(dim=1)
            scale = torch.clamp(max_abs / 127.0, min=1e-8)
            q = torch.clamp(torch.round(w / scale.unsqueeze(1)), -127, 127).to(torch.int8)
            layer.qweight.copy_(q)
            layer.scale.copy_(scale.to(torch.float16))
            if linear.bias is not None:
                layer.bias.copy_(linear.bias.detach().to(torch.float16))
        return layer

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        w = self.qweight.to(torch.float32) * self.scale.unsqueeze(1).to(torch.float32)
        bias = self.bias.to(torch.float32) if self.bias is not None else None
        out = F.linear(x.to(torch.float32), w, bias)
        return out.to(x.dtype)


def _to_device_map(device: Union[str, torch.device]):
    if isinstance(device, torch.device):
        device_str = str(device)
    else:
        device_str = str(device)

    if device_str.startswith("cuda"):
        return {"": device_str}
    return {"": "cpu"}


def _dir_size_bytes(path: Path) -> int:
    total = 0
    for root, _, files in os.walk(path):
        for name in files:
            total += (Path(root) / name).stat().st_size
    return total


def _swap_linears_to_int8(module: nn.Module):
    for name, child in list(module.named_children()):
        if isinstance(child, nn.Linear):
            setattr(module, name, Int8Linear.from_linear(child))
        else:
            _swap_linears_to_int8(child)

def _load_or_build_int8_fallback(device: Union[str, torch.device]) -> nn.Module:
    ARTIFACT_DIR.mkdir(parents=True, exist_ok=True)
    state_path = ARTIFACT_DIR / "fallback_int8_state.pt"
    cfg_path = ARTIFACT_DIR / "config.json"

    model = AutoModelForCausalLM.from_pretrained(
        MODEL_ID,
        dtype=torch.float16,
        device_map="cpu",
        low_cpu_mem_usage=True,
    )

    _swap_linears_to_int8(model)

    if state_path.exists():
        state = torch.load(state_path, map_location="cpu")
        model.load_state_dict(state, strict=True)
    else:
        torch.save(model.state_dict(), state_path)
        model.config.save_pretrained(ARTIFACT_DIR)

    if not cfg_path.exists():
        model.config.save_pretrained(ARTIFACT_DIR)

    if isinstance(device, torch.device):
        move_device = device
    else:
        move_device = torch.device(device)
    model.to(move_device)
    model.eval()

    size_bytes = _dir_size_bytes(ARTIFACT_DIR)
    SIZE_METADATA_PATH.write_text(json.dumps({"size_bytes": size_bytes}), encoding="utf-8")
    return model


def load_model(device="cuda"):
    """Load quantized Qwen3-4B model.

    Primary path: bitsandbytes 4-bit NF4 + double quantization.
    Fallback path: per-channel int8 linear quantization.
    """
    model = _load_or_build_bnb(device)
    if model is not None:
        return model
    return _load_or_build_int8_fallback(device)


def get_compressed_size() -> int:
    """Return on-disk compressed size in bytes.

    If metadata is missing, compute size from current artifact directory.
    """
    if SIZE_METADATA_PATH.exists():
        try:
            data = json.loads(SIZE_METADATA_PATH.read_text(encoding="utf-8"))
            size_bytes = int(data["size_bytes"])
            if size_bytes > 0:
                return size_bytes
        except Exception:
            pass

    if ARTIFACT_DIR.exists():
        size_bytes = _dir_size_bytes(ARTIFACT_DIR)
        if size_bytes > 0:
            SIZE_METADATA_PATH.write_text(json.dumps({"size_bytes": size_bytes}), encoding="utf-8")
            return size_bytes

    # Build artifacts on CPU if size is requested before model load.
    _ = load_model(device="cpu")
    size_bytes = _dir_size_bytes(ARTIFACT_DIR)
    SIZE_METADATA_PATH.write_text(json.dumps({"size_bytes": size_bytes}), encoding="utf-8")
    return size_bytes
