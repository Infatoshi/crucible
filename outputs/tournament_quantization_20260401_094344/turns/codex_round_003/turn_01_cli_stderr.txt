"""Qwen3-4B: AWQ 4-bit for linear layers + INT4 per-group for embeddings/lm_head."""

import torch
import torch.nn as nn
from awq import AutoAWQForCausalLM

MODEL_ID = "Qwen/Qwen3-4B"
AWQ_DIR = "./qwen3-4b-awq"
EMBED_GROUP_SIZE = 128


def quantize_to_int4(weight, group_size=EMBED_GROUP_SIZE):
    """Quantize weight to symmetric INT4 with per-group scales. Returns packed uint8."""
    rows, cols = weight.shape
    assert cols % group_size == 0
    num_groups = cols // group_size

    grouped = weight.reshape(rows, num_groups, group_size)
    scales = grouped.abs().amax(dim=-1, keepdim=True) / 7.0
    scales = scales.clamp(min=1e-8)

    quant = (grouped / scales).round().clamp(-8, 7)
    quant = (quant + 8).to(torch.uint8)
    quant = quant.reshape(rows, cols)

    packed = quant[:, 0::2] | (quant[:, 1::2] << 4)

    return packed, scales.squeeze(-1).half()


def dequantize_int4(packed, scales, hidden_size, group_size=EMBED_GROUP_SIZE):
    """Unpack and dequantize INT4 packed weights."""
    dtype = scales.dtype
    low = (packed & 0xF).to(dtype) - 8.0
    high = ((packed >> 4) & 0xF).to(dtype) - 8.0
    unpacked = torch.stack([low, high], dim=-1).reshape(*packed.shape[:-1], -1)

    num_groups = hidden_size // group_size
    unpacked = unpacked.reshape(*unpacked.shape[:-1], num_groups, group_size)
    result = unpacked * scales.unsqueeze(-1) if scales.dim() < unpacked.dim() else unpacked * scales[..., :, None]

    return result.reshape(*packed.shape[:-1], hidden_size)


class INT4Embedding(nn.Module):
    def __init__(self, packed_weight, scales, hidden_size):
        super().__init__()
        self.register_buffer('packed_weight', packed_weight)
        self.register_buffer('scales', scales)
        self.hidden_size = hidden_size

    def forward(self, input_ids):
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        return dequantize_int4(packed, scales, self.hidden_size)


class INT4LMHead(nn.Module):
    def __init__(self, packed_weight, scales, hidden_size, vocab_size):
        super().__init__()
        self.register_buffer('packed_weight', packed_weight)
        self.register_buffer('scales', scales)
        self.hidden_size = hidden_size
        self.vocab_size = vocab_size

    def forward(self, x):
        orig_shape = x.shape[:-1]
        x_flat = x.reshape(-1, x.shape[-1])
        dtype = x.dtype

        chunk_size = 32768
        outputs = []
        for i in range(0, self.vocab_size, chunk_size):
            end = min(i + chunk_size, self.vocab_size)
            w_deq = dequantize_int4(
                self.packed_weight[i:end], self.scales[i:end], self.hidden_size
            )
            outputs.append(x_flat @ w_deq.to(dtype).T)

        result = torch.cat(outputs, dim=-1)
        return result.reshape(*orig_shape, self.vocab_size)


def load_model(device="cuda"):
    """Load AWQ 4-bit quantized model with INT4 embeddings."""
    model = AutoAWQForCausalLM.from_quantized(AWQ_DIR, fuse_layers=False)
    model = model.model  # unwrap
    model.eval()

    # Quantize embedding to INT4 per-group
    embed_weight = model.model.embed_tokens.weight.data.cpu().float()
    packed, scales = quantize_to_int4(embed_weight)
    del embed_weight

    hidden_size = model.config.hidden_size
    vocab_size = model.config.vocab_size

    model.model.embed_tokens = INT4Embedding(
        packed.to(device), scales.to(device), hidden_size
    )
    model.lm_head = INT4LMHead(
        packed.to(device), scales.to(device), hidden_size, vocab_size
    )

    return model


def get_compressed_size():
    """Compressed size: AWQ 4-bit linears + INT4 embeddings + fp16 norms."""
    hidden = 2560
    layers = 36
    heads = 32
    kv_heads = 8
    intermediate = 9728
    vocab = 151936
    head_dim = hidden // heads

    # AWQ 4-bit linear layers
    q_params = hidden * heads * head_dim
    k_params = hidden * kv_heads * head_dim
    v_params = hidden * kv_heads * head_dim
    o_params = heads * head_dim * hidden
    attn_params = q_params + k_params + v_params + o_params
    mlp_params = 3 * hidden * intermediate
    total_quantized = (attn_params + mlp_params) * layers

    # AWQ: 4 bits per weight + fp16 scale per group + int4 zero per group
    group_size = 128
    bytes_per_weight = 0.5  # 4 bits packed
    scale_per_weight = 2.0 / group_size  # fp16 scale per group
    zero_per_weight = 0.5 / group_size  # int4 zero packed in int32
    awq_bytes = int(total_quantized * (bytes_per_weight + scale_per_weight + zero_per_weight))

    # INT4 embedding per-group (group_size=128)
    embed_params = vocab * hidden
    int4_embed_bytes = embed_params // 2
    num_groups = vocab * (hidden // EMBED_GROUP_SIZE)
    embed_scale_bytes = num_groups * 2
    embed_total = int4_embed_bytes + embed_scale_bytes

    # FP16 norms
    norm_params = layers * (2 * hidden + 2 * head_dim) + hidden
    norm_bytes = norm_params * 2

    return awq_bytes + embed_total + norm_bytes

def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

GROUP_SIZE = 64


# Model constants used by get_compressed_size().

NUM_LAYERS = 36

NUM_KV_HEADS = 8
INTERMEDIATE_SIZE = 9728

def _pack_nibbles(values: torch.Tensor) -> torch.Tensor:
    \"\"\"Pack uint4 values from last dim into uint8 (two values per byte).\"\"\"
    if values.dtype "'!= torch.uint8:
        raise TypeError("values must be torch.uint8")
    if values.shape[-1] % 2 != 0:
        raise ValueError("last dimension must be even for nibble packing")
    return values[..., 0::2] | (values[..., 1::2] << 4)


def _unpack_nibbles(packed: torch.Tensor) -> torch.Tensor:
    """Unpack uint8 nibbles into uint4 values on the last dim."""
    if packed.dtype != torch.uint8:
        raise TypeError("packed must be torch.uint8")
    low = packed & 0x0F
    high = (packed >> 4) & 0x0F
    return torch.stack((low, high), dim=-1).reshape(*packed.shape[:-1], packed.shape[-1] * 2)


def quantize_int4_asym_group(weight: torch.Tensor, group_size: int = GROUP_SIZE) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """Asymmetric per-group INT4 quantization.

    Returns:
    - packed_weight: uint8 with 2x int4 packed
    - scales: fp16, shape [rows, num_groups]
    - packed_zeros: uint8 with 2x uint4 packed zero-points
    """
    if weight.ndim != 2:
        raise ValueError("weight must be 2D [rows, cols]")

    rows, cols = weight.shape
    if cols % group_size != 0:
        raise ValueError("cols must be divisible by group_size")

    num_groups = cols // group_size
    grouped = weight.reshape(rows, num_groups, group_size)

    w_min = grouped.amin(dim=-1, keepdim=True)
    w_max = grouped.amax(dim=-1, keepdim=True)
    scales = ((w_max - w_min) / 15.0).clamp(min=1e-8)

    zeros = (-w_min / scales).round().clamp(0, 15).to(torch.uint8).squeeze(-1)
    quant = (grouped / scales + zeros.unsqueeze(-1).to(grouped.dtype)).round().clamp(0, 15).to(torch.uint8)

    quant = quant.reshape(rows, cols)
    packed_weight = _pack_nibbles(quant)

    if num_groups % 2 != 0:
        raise ValueError("num_groups must be even for packed zero-points")
    packed_zeros = _pack_nibbles(zeros)

    return packed_weight, scales.squeeze(-1).half(), packed_zeros


def dequantize_int4_asym_group(
    packed_weight: torch.Tensor,
    scales: torch.Tensor,
    packed_zeros: torch.Tensor,
    hidden_size: int,
    group_size: int = GROUP_SIZE,
) -> torch.Tensor:
    """Dequantize asymmetric per-group INT4 weights."""
    quant = _unpack_nibbles(packed_weight)
    zeros = _unpack_nibbles(packed_zeros)

    if quant.shape[-1] != hidden_size:
        raise ValueError("hidden_size mismatch with unpacked weights")

    dtype = scales.dtype
    num_groups = hidden_size // group_size

    quant = quant.reshape(*quant.shape[:-1], num_groups, group_size).to(dtype)
    zeros = zeros.to(dtype)
    dequant = (quant - zeros.unsqueeze(-1)) * scales.unsqueeze(-1)
    return dequant.reshape(*packed_weight.shape[:-1], hidden_size)


class INT4Embedding(nn.Module):
    def __init__(self, packed_weight: torch.Tensor, scales: torch.Tensor, packed_zeros: torch.Tensor, hidden_size: int):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("packed_zeros", packed_zeros)
        self.hidden_size = hidden_size

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        zeros = self.packed_zeros[input_ids]
        return dequantize_int4_asym_group(packed, scales, zeros, self.hidden_size)

def _pack_nibbles(values: torch.Tensor) -> torch.Tensor:
    """Pack uint4 values from last dim into uint8 (two values per byte)."""
    if values.dtype != torch.uint8:
        raise TypeError("values must be torch.uint8")
    if values.shape[-1] % 2 != 0:
        raise ValueError("last dimension must be even for nibble packing")
    return values[..., 0::2] | (values[..., 1::2] << 4)


def _unpack_nibbles(packed: torch.Tensor) -> torch.Tensor:
    """Unpack uint8 nibbles into uint4 values on the last dim."""
    if packed.dtype != torch.uint8:
        raise TypeError("packed must be torch.uint8")
    low = packed & 0x0F
    high = (packed >> 4) & 0x0F
    return torch.stack((low, high), dim=-1).reshape(*packed.shape[:-1], packed.shape[-1] * 2)


def quantize_int4_asym_group(weight: torch.Tensor, group_size: int = GROUP_SIZE) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """Asymmetric per-group INT4 quantization.

    Returns:
    - packed_weight: uint8 with 2x int4 packed
    - scales: fp16, shape [rows, num_groups]
    - packed_zeros: uint8 with 2x uint4 packed zero-points
    """
    if weight.ndim != 2:
        raise ValueError("weight must be 2D [rows, cols]")

    rows, cols = weight.shape
    if cols % group_size != 0:
        raise ValueError("cols must be divisible by group_size")

    num_groups = cols // group_size
    grouped = weight.reshape(rows, num_groups, group_size)

    w_min = grouped.amin(dim=-1, keepdim=True)
    w_max = grouped.amax(dim=-1, keepdim=True)
    scales = ((w_max - w_min) / 15.0).clamp(min=1e-8)

    zeros = (-w_min / scales).round().clamp(0, 15).to(torch.uint8).squeeze(-1)
    quant = (grouped / scales + zeros.unsqueeze(-1).to(grouped.dtype)).round().clamp(0, 15).to(torch.uint8)

    quant = quant.reshape(rows, cols)
    packed_weight = _pack_nibbles(quant)

    if num_groups % 2 != 0:
        raise ValueError("num_groups must be even for packed zero-points")
    packed_zeros = _pack_nibbles(zeros)

    return packed_weight, scales.squeeze(-1).half(), packed_zeros


def dequantize_int4_asym_group(
    packed_weight: torch.Tensor,
    scales: torch.Tensor,
    packed_zeros: torch.Tensor,
    hidden_size: int,
    group_size: int = GROUP_SIZE,
) -> torch.Tensor:
    """Dequantize asymmetric per-group INT4 weights."""
    quant = _unpack_nibbles(packed_weight)
    zeros = _unpack_nibbles(packed_zeros)

    if quant.shape[-1] != hidden_size:
        raise ValueError("hidden_size mismatch with unpacked weights")

    dtype = scales.dtype
    num_groups = hidden_size // group_size

    quant = quant.reshape(*quant.shape[:-1], num_groups, group_size).to(dtype)
    zeros = zeros.to(dtype)
    dequant = (quant - zeros.unsqueeze(-1)) * scales.unsqueeze(-1)
    return dequant.reshape(*packed_weight.shape[:-1], hidden_size)


class INT4Embedding(nn.Module):
    def __init__(self, packed_weight: torch.Tensor, scales: torch.Tensor, packed_zeros: torch.Tensor, hidden_size: int):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("packed_zeros", packed_zeros)
        self.hidden_size = hidden_size

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        zeros = self.packed_zeros[input_ids]
        return dequantize_int4_asym_group(packed, scales, zeros, self.hidden_size)
