def load_module(name, path):

def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

"""Qwen3-4B: NF4 for linear layers + INT4 per-group for embeddings/lm_head."""

import torch
import torch.nn as nn
from transformers import AutoModelForCausalLM, BitsAndBytesConfig

MODEL_ID = "Qwen/Qwen3-4B"
EMBED_GROUP_SIZE = 128


def quantize_to_int4(weight, group_size=EMBED_GROUP_SIZE):
    """Quantize weight to symmetric INT4 with per-group scales. Returns packed uint8."""
    rows, cols = weight.shape
    assert cols % group_size == 0
    num_groups = cols // group_size

    grouped = weight.reshape(rows, num_groups, group_size)
    scales = grouped.abs().amax(dim=-1, keepdim=True) / 7.0
    scales = scales.clamp(min=1e-8)

    quant = (grouped / scales).round().clamp(-8, 7)
    quant = (quant + 8).to(torch.uint8)  # shift to [0, 15]
    quant = quant.reshape(rows, cols)

    # Pack two 4-bit values per uint8 byte
    packed = quant[:, 0::2] | (quant[:, 1::2] << 4)  # [rows, cols//2]

    return packed, scales.squeeze(-1).half()  # [rows, num_groups]


def dequantize_int4(packed, scales, hidden_size, group_size=EMBED_GROUP_SIZE):
    """Unpack and dequantize INT4 packed weights."""
    dtype = scales.dtype
    low = (packed & 0xF).to(dtype) - 8.0
    high = ((packed >> 4) & 0xF).to(dtype) - 8.0
    unpacked = torch.stack([low, high], dim=-1).reshape(*packed.shape[:-1], -1)

    num_groups = hidden_size // group_size
    unpacked = unpacked.reshape(*unpacked.shape[:-1], num_groups, group_size)
    result = unpacked * scales.unsqueeze(-1) if scales.dim() < unpacked.dim() else unpacked * scales[..., :, None]

    return result.reshape(*packed.shape[:-1], hidden_size)


class INT4Embedding(nn.Module):
    def __init__(self, packed_weight, scales, hidden_size):
        super().__init__()
        self.register_buffer('packed_weight', packed_weight)
        self.register_buffer('scales', scales)
        self.hidden_size = hidden_size

    def forward(self, input_ids):
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        return dequantize_int4(packed, scales, self.hidden_size)


class INT4LMHead(nn.Module):
    def __init__(self, packed_weight, scales, hidden_size, vocab_size):
        super().__init__()
        self.register_buffer('packed_weight', packed_weight)
        self.register_buffer('scales', scales)
        self.hidden_size = hidden_size
        self.vocab_size = vocab_size

    def forward(self, x):
        orig_shape = x.shape[:-1]
        x_flat = x.reshape(-1, x.shape[-1])
        dtype = x.dtype

        chunk_size = 32768
        outputs = []
        for i in range(0, self.vocab_size, chunk_size):
            end = min(i + chunk_size, self.vocab_size)
            w_deq = dequantize_int4(
                self.packed_weight[i:end], self.scales[i:end], self.hidden_size
            )
            outputs.append(x_flat @ w_deq.to(dtype).T)

        result = torch.cat(outputs, dim=-1)
        return result.reshape(*orig_shape, self.vocab_size)


def load_model(device="cuda"):
    """Load model with NF4 linear layers and INT4 embeddings."""
    bnb_config = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_compute_dtype=torch.float16,
        bnb_4bit_use_double_quant=True,
    )
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_ID,
        quantization_config=bnb_config,
        device_map=device,
    )
    model.eval()

    # Quantize embedding to INT4 per-group
    embed_weight = model.model.embed_tokens.weight.data.cpu().float()
    packed, scales = quantize_to_int4(embed_weight)
    del embed_weight

    hidden_size = model.config.hidden_size
    vocab_size = model.config.vocab_size

    model.model.embed_tokens = INT4Embedding(
        packed.to(device), scales.to(device), hidden_size
    )
    model.lm_head = INT4LMHead(
        packed.to(device), scales.to(device), hidden_size, vocab_size
    )

    return model
