def _pack_nibbles(values: torch.Tensor) -> torch.Tensor:
    """Pack uint4 values from last dim into uint8 (two values per byte)."""
    if values.dtype != torch.uint8:
        raise TypeError("values must be torch.uint8")
    if values.shape[-1] % 2 != 0:
        raise ValueError("last dimension must be even for nibble packing")
    return values[..., 0::2] | (values[..., 1::2] << 4)


def _unpack_nibbles(packed: torch.Tensor) -> torch.Tensor:
    """Unpack uint8 nibbles into uint4 values on the last dim."""
    if packed.dtype != torch.uint8:
        raise TypeError("packed must be torch.uint8")
    low = packed & 0x0F
    high = (packed >> 4) & 0x0F
    return torch.stack((low, high), dim=-1).reshape(*packed.shape[:-1], packed.shape[-1] * 2)


def quantize_int4_asym_group(weight: torch.Tensor, group_size: int = GROUP_SIZE) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """Asymmetric per-group INT4 quantization."""
    if weight.ndim != 2:
        raise ValueError("weight must be 2D [rows, cols]")

    rows, cols = weight.shape
    if cols % group_size != 0:
        raise ValueError("cols must be divisible by group_size")

    num_groups = cols // group_size
    grouped = weight.reshape(rows, num_groups, group_size)

    w_min = grouped.amin(dim=-1, keepdim=True)
    w_max = grouped.amax(dim=-1, keepdim=True)
    scales = ((w_max - w_min) / 15.0).clamp(min=1e-8)

    zeros = (-w_min / scales).round().clamp(0, 15).to(torch.uint8).squeeze(-1)
    quant = (grouped / scales + zeros.unsqueeze(-1).to(grouped.dtype)).round().clamp(0, 15).to(torch.uint8)

    quant = quant.reshape(rows, cols)
    packed_weight = _pack_nibbles(quant)

    if num_groups % 2 != 0:
        raise ValueError("num_groups must be even for packed zero-points")
    packed_zeros = _pack_nibbles(zeros)

    return packed_weight, scales.squeeze(-1).half(), packed_zeros


def dequantize_int4_asym_group(
    packed_weight: torch.Tensor,
    scales: torch.Tensor,
    packed_zeros: torch.Tensor,
    hidden_size: int,
    group_size: int = GROUP_SIZE,
) -> torch.Tensor:
    """Dequantize asymmetric per-group INT4 weights."""
    quant = _unpack_nibbles(packed_weight)
    zeros = _unpack_nibbles(packed_zeros)

    if quant.shape[-1] != hidden_size:
        raise ValueError("hidden_size mismatch with unpacked weights")

    dtype = scales.dtype
    num_groups = hidden_size // group_size

    quant = quant.reshape(*quant.shape[:-1], num_groups, group_size).to(dtype)
    zeros = zeros.to(dtype)
    dequant = (quant - zeros.unsqueeze(-1)) * scales.unsqueeze(-1)
    return dequant.reshape(*packed_weight.shape[:-1], hidden_size)


def _build_exact_token_cache(tokenizer, max_tokens: int = MAX_EXACT_TOKENS) -> torch.Tensor:
    token_set = set()
    for text in CALIB_TEXTS:
        ids = tokenizer.encode(text, add_special_tokens=False)
        token_set.update(ids)
    if not token_set:
        return torch.empty(0, dtype=torch.long)
    token_ids = torch.tensor(sorted(token_set), dtype=torch.long)
    if token_ids.numel() > max_tokens:
        token_ids = token_ids[:max_tokens]
    return token_ids


class INT4Embedding(nn.Module):
    def __init__(self, packed_weight, scales, packed_zeros, hidden_size,
                 exact_token_ids, exact_rows, vocab_size):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("packed_zeros", packed_zeros)
        self.register_buffer("exact_token_ids", exact_token_ids)
        self.register_buffer("exact_rows", exact_rows)
        self.hidden_size = hidden_size

        lookup = torch.full((vocab_size,), -1, dtype=torch.int32, device=packed_weight.device)
        if exact_token_ids.numel() > 0:
            lookup[exact_token_ids] = torch.arange(
                exact_token_ids.numel(), dtype=torch.int32, device=packed_weight.device
            )
        self.register_buffer("exact_lookup", lookup)

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        zeros = self.packed_zeros[input_ids]
        x = dequantize_int4_asym_group(packed, scales, zeros, self.hidden_size)

        if self.exact_rows.numel() > 0:
            idx = self.exact_lookup[input_ids]
            mask = idx >= 0
            if mask.any():
                x[mask] = self.exact_rows[idx[mask]].to(x.dtype)
        return x


class INT4LMHead(nn.Module):
    def __init__(self, packed_weight, scales, packed_zeros, hidden_size,
                 vocab_size, exact_token_ids, exact_rows):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("packed_zeros", packed_zeros)
        self.register_buffer("exact_token_ids", exact_token_ids)
        self.register_buffer("exact_rows", exact_rows)
        self.hidden_size = hidden_size
        self.vocab_size = vocab_size

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        orig_shape = x.shape[:-1]
        x_flat = x.reshape(-1, x.shape[-1])
        x_mat = x_flat.float()

        chunk_size = 16384
        outputs = []
        for i in range(0, self.vocab_size, chunk_size):
            end = min(i + chunk_size, self.vocab_size)
            w_deq = dequantize_int4_asym_group(
                self.packed_weight[i:end],
                self.scales[i:end],
                self.packed_zeros[i:end],
                self.hidden_size,
            )
            outputs.append(x_mat @ w_deq.float().T)

        logits = torch.cat(outputs, dim=-1)

        if self.exact_rows.numel() > 0:
            exact_logits = x_mat @ self.exact_rows.float().T
            logits[:, self.exact_token_ids] = exact_logits

        return logits.reshape(*orig_shape, self.vocab_size) * LOGIT_SCALE

        torch.backends.cudnn.allow_tf32 = False

    source = _resolve_awq_source()
    wrapped = AutoAWQForCausalLM.from_quantized(
        source,
        fuse_layers=False,
        trust_remote_code=True,
    )
    model = wrapped.model if hasattr(wrapped, "model") else wrapped
    model.eval()

    target_device = torch.device(device) if not isinstance(device, torch.device) else device

    with torch.no_grad():
        tokenizer = AutoTokenizer.from_pretrained(MODEL_ID, trust_remote_code=True)
        exact_token_ids_cpu = _build_exact_token_cache(tokenizer)

        embed_weight = model.model.embed_tokens.weight.detach().to("cpu", dtype=torch.float32)
        packed_weight, scales, packed_zeros = quantize_int4_asym_group(
            embed_weight, group_size=GROUP_SIZE
        )

        if exact_token_ids_cpu.numel() > 0:
            exact_rows_cpu = embed_weight[exact_token_ids_cpu].half()
        else:
            exact_rows_cpu = torch.empty(0, embed_weight.shape[1], dtype=torch.float16)

    packed_weight = packed_weight.to(target_device)
    scales = scales.to(target_device)
    packed_zeros = packed_zeros.to(target_device)
    exact_token_ids = exact_token_ids_cpu.to(target_device, dtype=torch.long)
    exact_rows = exact_rows_cpu.to(target_device)

    hidden_size = model.config.hidden_size
    vocab_size = model.config.vocab_size

    model.model.embed_tokens = INT4Embedding(
        packed_weight=packed_weight,
        scales=scales,
        packed_zeros=packed_zeros,
        hidden_size=hidden_size,
        exact_token_ids=exact_token_ids,
        exact_rows=exact_rows,
        vocab_size=vocab_size,
    )
    model.lm_head = INT4LMHead(
        packed_weight=packed_weight,
        scales=scales,
        packed_zeros=packed_zeros,
        hidden_size=hidden_size,
        vocab_size=vocab_size,
        exact_token_ids=exact_token_ids,
        exact_rows=exact_rows,
    )

    return model


def get_compressed_size() -> int:
    """Honest compressed size in bytes for AWQ4 linears + INT4 embedding/lm_head + exact-token cache."""
    head_dim = HIDDEN_SIZE // NUM_ATTN_HEADS

    # Attention + MLP linear weights (quantized by AWQ).
    q_params = HIDDEN_SIZE * NUM_ATTN_HEADS * head_dim
    k_params = HIDDEN_SIZE * NUM_KV_HEADS * head_dim
    v_params = HIDDEN_SIZE * NUM_KV_HEADS * head_dim
    o_params = NUM_ATTN_HEADS * head_dim * HIDDEN_SIZE
    attn_params = q_params + k_params + v_params + o_params
    mlp_params = 3 * HIDDEN_SIZE * INTERMEDIATE_SIZE
    total_linear_params = (attn_params + mlp_params) * NUM_LAYERS

    # AWQ storage: int4 weights + fp16 scales + packed int4 zeros per group.
    awq_bytes = int(total_linear_params * (0.5 + (2.0 / AWQ_GROUP_SIZE) + (0.5 / AWQ_GROUP_SIZE)))

    # Shared embedding/lm_head stored once as asymmetric int4 group quantized tensors.
    embed_params = VOCAB_SIZE * HIDDEN_SIZE

MAX_SEQ_LEN = 512  # Truncate to this length for evaluation


def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

    device = torch.device("cuda:0")

    total_eval_tokens = sum(len(ids) for ids in input_ids_list)

    ref = load_module("reference", "reference.py")
    ref_model = ref.load_model(device=device)
    ref_ppl = compute_perplexity(ref_model, input_ids_list, device)
    del ref_model, ref
    gc.collect()
    torch.cuda.empty_cache()

    # ---- Phase 2: Solution model ----
    sol = load_module("solution", "solution.py")

    if not hasattr(sol, "get_compressed_size"):
        print(json.dumps({
            "fitness": 0.0,
            "error": "Solution must define get_compressed_size() -> int"
        }))
        sys.exit(0)

    # Load solution model
    if hasattr(sol, "load_model"):
        sol_model = sol.load_model(device=device)
    elif hasattr(sol, "Model"):
        sol_model = sol.Model()
        if hasattr(sol_model, "to"):
            sol_model = sol_model.to(device)
        sol_model.eval()
    else:
        print(json.dumps({
            "fitness": 0.0,
            "error": "Solution must define load_model(device) or Model class"
        }))
        sys.exit(0)

    sol_bytes = sol.get_compressed_size()

    if not isinstance(sol_bytes, (int, float)) or sol_bytes <= 0:
        print(json.dumps({
            "fitness": 0.0,
            "error": f"get_compressed_size() must return positive number, got {sol_bytes}"
        }))
        sys.exit(0)

    sol_bytes = int(sol_bytes)

    if sol_bytes >= REF_BYTES:
        print(json.dumps({
            "fitness": 0.0,
            "error": f"No compression: sol_bytes={sol_bytes} >= ref_bytes={REF_BYTES}",
        }))
        sys.exit(0)

    compression_ratio = REF_BYTES / sol_bytes

    # Sanity: compression > 32x is suspicious for a 4B model
    if compression_ratio > 32:
        print(json.dumps({
            "fitness": 0.0,
            "error": f"Compression ratio {compression_ratio:.1f}x exceeds 32x cap -- likely dishonest size claim",
        }))
        sys.exit(0)

    sol_ppl = compute_perplexity(sol_model, input_ids_list, device)

    # Check for degenerate output
    if math.isnan(sol_ppl) or math.isinf(sol_ppl) or sol_ppl > 1e6:
        print(json.dumps({
            "fitness": 0.0,
            "error": f"Degenerate perplexity: {sol_ppl}",
            "ref_ppl": round(ref_ppl, 4),
        }))
        sys.exit(0)

    # Quality = exp(-max(0, normalized ppl increase))
    # Normalized by ref_ppl so the scale is relative, not absolute
    delta_ppl_norm = (sol_ppl - ref_ppl) / ref_ppl
    quality = math.exp(-max(0.0, delta_ppl_norm))
    quality = min(quality, 1.0)  # Cap at 1.0

    # Fitness = quality * log2(compression_ratio)
    fitness = quality * math.log2(max(compression_ratio, 1.01))

    del sol_model, sol
    gc.collect()
    torch.cuda.empty_cache()

    print(json.dumps({
        "fitness": round(fitness, 6),
        "quality": round(quality, 6),
        "compression_ratio": round(compression_ratio, 4),
        "ref_ppl": round(ref_ppl, 4),
        "sol_ppl": round(sol_ppl, 4),
        "delta_ppl_pct": round(delta_ppl_norm * 100, 2),
        "ref_bytes": REF_BYTES,
        "sol_bytes": sol_bytes,
        "eval_tokens": total_eval_tokens,
    }))

except Exception as e:

    print(json.dumps({
        "fitness": 0.0,
        "error": str(e),
        "traceback": traceback.format_exc()[:2000],
    }))
