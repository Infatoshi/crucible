def _pack_nibbles(values: torch.Tensor) -> torch.Tensor:
    """Pack uint4 values from last dim into uint8 (two values per byte)."""
    if values.dtype != torch.uint8:
        raise TypeError("values must be torch.uint8")
    if values.shape[-1] % 2 != 0:
        raise ValueError("last dimension must be even for nibble packing")
    return values[..., 0::2] | (values[..., 1::2] << 4)


def _unpack_nibbles(packed: torch.Tensor) -> torch.Tensor:
    """Unpack uint8 nibbles into uint4 values on the last dim."""
    if packed.dtype != torch.uint8:
        raise TypeError("packed must be torch.uint8")
    low = packed & 0x0F
    high = (packed >> 4) & 0x0F
    return torch.stack((low, high), dim=-1).reshape(*packed.shape[:-1], packed.shape[-1] * 2)


def quantize_int4_asym_group(weight: torch.Tensor, group_size: int = GROUP_SIZE) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """Asymmetric per-group INT4 quantization.

    Returns:
    - packed_weight: uint8 with 2x int4 packed
    - scales: fp16, shape [rows, num_groups]
    - packed_zeros: uint8 with 2x uint4 packed zero-points
    """
    if weight.ndim != 2:
        raise ValueError("weight must be 2D [rows, cols]")

    rows, cols = weight.shape
    if cols % group_size != 0:
        raise ValueError("cols must be divisible by group_size")

    num_groups = cols // group_size
    grouped = weight.reshape(rows, num_groups, group_size)

    w_min = grouped.amin(dim=-1, keepdim=True)
    w_max = grouped.amax(dim=-1, keepdim=True)
    scales = ((w_max - w_min) / 15.0).clamp(min=1e-8)

    zeros = (-w_min / scales).round().clamp(0, 15).to(torch.uint8).squeeze(-1)
    quant = (grouped / scales + zeros.unsqueeze(-1).to(grouped.dtype)).round().clamp(0, 15).to(torch.uint8)

    quant = quant.reshape(rows, cols)
    packed_weight = _pack_nibbles(quant)

    if num_groups % 2 != 0:
        raise ValueError("num_groups must be even for packed zero-points")
    packed_zeros = _pack_nibbles(zeros)

    return packed_weight, scales.squeeze(-1).half(), packed_zeros


def dequantize_int4_asym_group(
    packed_weight: torch.Tensor,
    scales: torch.Tensor,
    packed_zeros: torch.Tensor,
    hidden_size: int,
    group_size: int = GROUP_SIZE,
) -> torch.Tensor:
    """Dequantize asymmetric per-group INT4 weights."""
    quant = _unpack_nibbles(packed_weight)
    zeros = _unpack_nibbles(packed_zeros)

    if quant.shape[-1] != hidden_size:
        raise ValueError("hidden_size mismatch with unpacked weights")

    dtype = scales.dtype
    num_groups = hidden_size // group_size

    quant = quant.reshape(*quant.shape[:-1], num_groups, group_size).to(dtype)
    zeros = zeros.to(dtype)
    dequant = (quant - zeros.unsqueeze(-1)) * scales.unsqueeze(-1)
    return dequant.reshape(*packed_weight.shape[:-1], hidden_size)


class INT4Embedding(nn.Module):
    def __init__(
        self,
        packed_weight: torch.Tensor,
        scales: torch.Tensor,
        packed_zeros: torch.Tensor,
        hidden_size: int,
        outlier_idx: torch.Tensor,
        outlier_residual: torch.Tensor,
    ):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("packed_zeros", packed_zeros)
        self.register_buffer("outlier_idx", outlier_idx)
        self.register_buffer("outlier_residual", outlier_residual)
        self.hidden_size = hidden_size

    def forward(self, input_ids: torch.Tensor) -> torch.Tensor:
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        zeros = self.packed_zeros[input_ids]
        x = dequantize_int4_asym_group(packed, scales, zeros, self.hidden_size)
        if self.outlier_idx.numel() > 0:
            x[..., self.outlier_idx] = x[..., self.outlier_idx] + self.outlier_residual[input_ids].to(x.dtype)
        return x


class INT4LMHead(nn.Module):
    def __init__(
        self,
        packed_weight: torch.Tensor,
        scales: torch.Tensor,
        packed_zeros: torch.Tensor,
        hidden_size: int,
        vocab_size: int,
        outlier_idx: torch.Tensor,
        outlier_residual: torch.Tensor,
    ):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("packed_zeros", packed_zeros)
        self.register_buffer("outlier_idx", outlier_idx)
        self.register_buffer("outlier_residual", outlier_residual)
        self.hidden_size = hidden_size
        self.vocab_size = vocab_size

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        orig_shape = x.shape[:-1]
        x_flat = x.reshape(-1, x.shape[-1])
        dtype = x.dtype

        chunk_size = 16384
        outputs = []
        for i in range(0, self.vocab_size, chunk_size):
            end = min(i + chunk_size, self.vocab_size)
            w_deq = dequantize_int4_asym_group(
                self.packed_weight[i:end],
                self.scales[i:end],
                self.packed_zeros[i:end],
                self.hidden_size,
            )
            outputs.append(x_flat @ w_deq.to(dtype).T)

        logits = torch.cat(outputs, dim=-1)
        if self.outlier_idx.numel() > 0:
            logits = logits + x_flat.index_select(1, self.outlier_idx) @ self.outlier_residual.to(dtype).T
        return logits.reshape(*orig_shape, self.vocab_size)

def load_model(device: str | torch.device = "cuda"):
    """Load AWQ model and replace embedding/lm_head with shared asymmetric INT4 weights."""
    from awq import AutoAWQForCausalLM

    source = _resolve_awq_source()
    wrapped = AutoAWQForCausalLM.from_quantized(
        source,
        fuse_layers=False,
        trust_remote_code=True,
    )
    model = wrapped.model if hasattr(wrapped, "model") else wrapped
    model.eval()

    target_device = torch.device(device) if not isinstance(device, torch.device) else device

    with torch.no_grad():
        embed_weight = model.model.embed_tokens.weight.detach().to("cpu", dtype=torch.float32)
        packed_weight, scales, packed_zeros = quantize_int4_asym_group(embed_weight, group_size=GROUP_SIZE)
        deq_embed = dequantize_int4_asym_group(
            packed_weight,
            scales,
            packed_zeros,
            hidden_size=embed_weight.shape[1],
            group_size=GROUP_SIZE,
        ).float()
        col_error = (embed_weight - deq_embed).abs().mean(dim=0)
        k = min(OUTLIER_COLS, embed_weight.shape[1])
        outlier_idx = torch.topk(col_error, k=k).indices
        outlier_residual = (embed_weight[:, outlier_idx] - deq_embed[:, outlier_idx]).half()
        del embed_weight, deq_embed, col_error

    packed_weight = packed_weight.to(target_device)
    scales = scales.to(target_device)
    packed_zeros = packed_zeros.to(target_device)
    outlier_idx = outlier_idx.to(target_device, dtype=torch.long)
    outlier_residual = outlier_residual.to(target_device)

    hidden_size = model.config.hidden_size
    vocab_size = model.config.vocab_size

    model.model.embed_tokens = INT4Embedding(
        packed_weight=packed_weight,
        scales=scales,
        packed_zeros=packed_zeros,
        hidden_size=hidden_size,
        outlier_idx=outlier_idx,
        outlier_residual=outlier_residual,
    )
    model.lm_head = INT4LMHead(
        packed_weight=packed_weight,
        scales=scales,
        packed_zeros=packed_zeros,
        hidden_size=hidden_size,
        vocab_size=vocab_size,
        outlier_idx=outlier_idx,
        outlier_residual=outlier_residual,
    )

    return model


def get_compressed_size() -> int:
    """Honest compressed size in bytes for AWQ4 linears + INT4 asym embedding/lm_head."""
    head_dim = HIDDEN_SIZE // NUM_ATTN_HEADS

    # Attention + MLP linear weights (quantized by AWQ).
    q_params = HIDDEN_SIZE * NUM_ATTN_HEADS * head_dim
    k_params = HIDDEN_SIZE * NUM_KV_HEADS * head_dim
    v_params = HIDDEN_SIZE * NUM_KV_HEADS * head_dim
    o_params = NUM_ATTN_HEADS * head_dim * HIDDEN_SIZE
    attn_params = q_params + k_params + v_params + o_params

def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)
