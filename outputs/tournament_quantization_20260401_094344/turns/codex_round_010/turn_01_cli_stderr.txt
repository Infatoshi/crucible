    """Per-layer parameter counts and bytes (fp16) for mixed-precision planning."""
    model = AutoModelForCausalLM.from_pretrained(MODEL_ID, dtype=torch.float16)
    layers = {}
    for name, p in model.named_parameters():
        # Group by layer
        parts = name.split(".")
        if "layers" in parts:
            idx = parts.index("layers")
            group = f"layers.{parts[idx+1]}.{parts[idx+2]}"
        else:
            group = parts[0] if len(parts) > 1 else name
        if group not in layers:
            layers[group] = {"params": 0, "bytes": 0, "shapes": []}
        layers[group]["params"] += p.numel()
        layers[group]["bytes"] += p.numel() * p.element_size()
        layers[group]["shapes"].append((name, list(p.shape)))
    del model
    return layers

MAX_SEQ_LEN = 512  # Truncate to this length for evaluation


def load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

# ============================================================
# 3-bit packing: 8 values (0-7) -> 3 bytes
# ============================================================

def pack_3bit(values: torch.Tensor) -> torch.Tensor:
    """Pack uint8 values (0-7) into 3-bit packed format. Last dim must be divisible by 8."""
    assert values.dtype == torch.uint8
    orig_shape = values.shape
    n = orig_shape[-1]
    assert n % 8 == 0, f"Last dim {n} must be divisible by 8"

    flat = values.reshape(-1, 8)
    v = flat

    b0 = v[:, 0] | (v[:, 1] << 3) | ((v[:, 2] & 0x3) << 6)
    b1 = (v[:, 2] >> 2) | (v[:, 3] << 1) | (v[:, 4] << 4) | ((v[:, 5] & 0x1) << 7)
    b2 = (v[:, 5] >> 1) | (v[:, 6] << 2) | (v[:, 7] << 5)

    packed = torch.stack([b0, b1, b2], dim=-1)
    return packed.reshape(*orig_shape[:-1], n * 3 // 8)


def unpack_3bit(packed: torch.Tensor, n_values: int) -> torch.Tensor:
    """Unpack 3-bit packed format to uint8 values (0-7)."""
    orig_shape = packed.shape
    flat = packed.reshape(-1, 3)

    b0 = flat[:, 0].to(torch.int32)
    b1 = flat[:, 1].to(torch.int32)
    b2 = flat[:, 2].to(torch.int32)

    v0 = (b0 & 0x7).to(torch.uint8)
    v1 = ((b0 >> 3) & 0x7).to(torch.uint8)
    v2 = (((b0 >> 6) | (b1 << 2)) & 0x7).to(torch.uint8)
    v3 = ((b1 >> 1) & 0x7).to(torch.uint8)
    v4 = ((b1 >> 4) & 0x7).to(torch.uint8)
    v5 = (((b1 >> 7) | (b2 << 1)) & 0x7).to(torch.uint8)
    v6 = ((b2 >> 2) & 0x7).to(torch.uint8)
    v7 = ((b2 >> 5) & 0x7).to(torch.uint8)

    result = torch.stack([v0, v1, v2, v3, v4, v5, v6, v7], dim=-1)
    return result.reshape(*orig_shape[:-1], n_values)


# ============================================================
# 4-bit packing (for embedding/lm_head)
# ============================================================

def pack_nibbles(values: torch.Tensor) -> torch.Tensor:
    assert values.dtype == torch.uint8
    assert values.shape[-1] % 2 == 0
    return values[..., 0::2] | (values[..., 1::2] << 4)


def unpack_nibbles(packed: torch.Tensor) -> torch.Tensor:
    low = packed & 0x0F
    high = (packed >> 4) & 0x0F
    return torch.stack((low, high), dim=-1).reshape(*packed.shape[:-1], packed.shape[-1] * 2)

# GPTQ core algorithm
# ============================================================

@torch.no_grad()
def gptq_quantize_weight(W, H, bits=3, group_size=128, blocksize=128, percdamp=0.01):
    """GPTQ quantization of a weight matrix.

    Args:
        W: (out_features, in_features) weight matrix in float32
        H: (in_features, in_features) Hessian = X^T @ X / n_samples
        bits: quantization bits
        group_size: group size for per-group quantization
        blocksize: GPTQ block size

    Returns:
        Q: (out_features, in_features) quantized values as uint8 (0 to 2^bits-1)
        scales: (out_features, n_groups) fp16 scales
        zeros: (out_features, n_groups) uint8 zeros
    """
    device = W.device
    rows, cols = W.shape
    maxq = 2 ** bits - 1
    n_groups = cols // group_size

    # Handle dead columns
    dead = torch.diag(H) == 0
    H[dead, dead] = 1.0
    W[:, dead] = 0.0

    # Add dampening
    damp = percdamp * torch.mean(torch.diag(H))
    diag_idx = torch.arange(cols, device=device)
    H[diag_idx, diag_idx] += damp

    # Cholesky decomposition of H^{-1}
    try:
        H_chol = torch.linalg.cholesky(H)
        H_inv = torch.cholesky_inverse(H_chol)
        Hinv = torch.linalg.cholesky(H_inv, upper=True)
    except Exception:
        # Fallback: increase dampening and retry
        H[diag_idx, diag_idx] += 10 * damp
        try:
            H_chol = torch.linalg.cholesky(H)
            H_inv = torch.cholesky_inverse(H_chol)
            Hinv = torch.linalg.cholesky(H_inv, upper=True)
        except Exception:
            # Final fallback: use diagonal approximation (RTN)
            Hinv = torch.zeros(cols, cols, device=device, dtype=W.dtype)
            Hinv[diag_idx, diag_idx] = 1.0 / torch.sqrt(torch.diag(H))

    Q = torch.zeros_like(W, dtype=torch.uint8)
    scales = torch.zeros(rows, n_groups, device=device, dtype=torch.float32)
    zeros = torch.zeros(rows, n_groups, device=device, dtype=torch.float32)

    for i1 in range(0, cols, blocksize):
        i2 = min(i1 + blocksize, cols)
        count = i2 - i1

        W1 = W[:, i1:i2].clone()
        Err1 = torch.zeros_like(W1)
        Hinv1 = Hinv[i1:i2, i1:i2]

        for j in range(count):
            col = i1 + j
            g = col // group_size

            # Recompute scale/zero at group boundary
            if col % group_size == 0:
                g_start = col
                g_end = min(col + group_size, cols)
                w_group = W[:, g_start:g_end]
                w_min = w_group.amin(dim=1)
                w_max = w_group.amax(dim=1)
                sc = ((w_max - w_min) / maxq).clamp(min=1e-8)
                zp = (-w_min / sc).round().clamp(0, maxq)
                scales[:, g] = sc
                zeros[:, g] = zp

            w = W1[:, j]
            d = Hinv1[j, j]

            sc = scales[:, g]
            zp = zeros[:, g]

            # Quantize
            q = (w / sc + zp).round().clamp(0, maxq)
            Q[:, col] = q.to(torch.uint8)

            # Dequantize
            q_deq = (q - zp) * sc

            # Error and propagation within block
            err = (w - q_deq) / d
            W1[:, j + 1:] -= err.unsqueeze(1) * Hinv1[j, j + 1:count].unsqueeze(0)
            Err1[:, j] = err

        # Propagate error to remaining columns outside block
        W[:, i2:] -= Err1 @ Hinv[i1:i2, i2:]

    return Q, scales.half(), zeros.to(torch.uint8)

@torch.no_grad()
def collect_layer_inputs(model, tokenizer, calib_texts, device):
    """Capture inputs to the first transformer block (per-sample kwargs for varying seq lengths)."""
    inps = []
    all_kwargs = []

    layers = model.model.layers

    class CatchInput(nn.Module):
        def __init__(self, module):
            super().__init__()
            self.module = module

        def forward(self, *args, **kwargs):
            inps.append(args[0].detach())
            kw = {}
            for k, v in kwargs.items():
                if isinstance(v, torch.Tensor):
                    kw[k] = v.detach()
                elif isinstance(v, tuple):
                    kw[k] = tuple(
                        x.detach() if isinstance(x, torch.Tensor) else x for x in v
                    )
                else:
                    kw[k] = v
            all_kwargs.append(kw)
            raise ValueError("catch")

    layers[0] = CatchInput(layers[0])
    for text in calib_texts:
        ids = tokenizer(text, return_tensors="pt", truncation=True, max_length=512).input_ids.to(device)
        try:
            model(ids)
        except ValueError:
            pass
    layers[0] = layers[0].module

    return inps, all_kwargs


@torch.no_grad()
def quantize_layer_gptq(layer, inps, all_kwargs, bits=3, group_size=128, device="cuda"):
    """Quantize all linear modules in a transformer layer using GPTQ.

    Returns dict of quantized data and list of outputs for next layer.
    """
    # Identify all linear modules
    linears = {}
    for name, module in layer.named_modules():
        if isinstance(module, nn.Linear):
            linears[name] = module

    # Collect Hessians by hooking each linear
    hessians = {}
    n_samples_dict = {}
    hooks = []

    def make_hook(name):
        def hook_fn(module, inp, out):
            x = inp[0].detach().float()
            if x.ndim == 3:
                x = x.reshape(-1, x.shape[-1])
            n = x.shape[0]
            H = x.T @ x
            if name not in hessians:
                hessians[name] = H
                n_samples_dict[name] = n
            else:
                hessians[name] += H
                n_samples_dict[name] += n
        return hook_fn

    for name, module in linears.items():
        hooks.append(module.register_forward_hook(make_hook(name)))

    # Run inputs through layer (with per-sample kwargs)
    outs = []
    for inp, kw in zip(inps, all_kwargs):
        out = layer(inp, **kw)
        if isinstance(out, tuple):
            outs.append(out[0].detach())
        else:
            outs.append(out.detach())

    for h in hooks:
        h.remove()

    # Normalize Hessians
    for name in hessians:
        hessians[name] /= n_samples_dict[name]

    # GPTQ quantize each linear
    quantized = {}
    for name, module in linears.items():
        W = module.weight.data.float()
        H = hessians[name]

        Q, scales, zeros = gptq_quantize_weight(W, H, bits=bits, group_size=group_size)

        # Pack weights
        in_features = module.in_features
        out_features = module.out_features

        # Pad to multiple of 8 for 3-bit packing
        if in_features % 8 != 0:

        if in_features % 8 != 0:
            pad = 8 - (in_features % 8)
            Q = F.pad(Q, (0, pad), value=0)
            packed = pack_3bit(Q)
        else:
            packed = pack_3bit(Q)

        bias = module.bias.data.half() if module.bias is not None else None

        quantized[name] = {
            "packed_weight": packed.cpu(),
            "scales": scales.cpu(),
            "zeros": zeros.cpu(),
            "bias": bias.cpu() if bias is not None else None,
            "in_features": in_features,
            "out_features": out_features,
            "bits": bits,
        }

        # Update module weight with dequantized values for accurate output propagation
        Q_unpacked = unpack_3bit(packed.to(device), in_features if in_features % 8 == 0 else in_features + (8 - in_features % 8))
        Q_unpacked = Q_unpacked[:, :in_features]
        n_groups = in_features // group_size
        q_grouped = Q_unpacked.reshape(out_features, n_groups, group_size).float()
        z = zeros[:, :n_groups].float().to(device)
        s = scales[:, :n_groups].float().to(device)
        w_deq = (q_grouped - z.unsqueeze(-1)) * s.unsqueeze(-1)
        w_deq = w_deq.reshape(out_features, in_features)
        module.weight.data = w_deq.half()

    # Re-run inputs through quantized layer for accurate propagation
    outs = []
    for inp, kw in zip(inps, all_kwargs):
        out = layer(inp, **kw)
        if isinstance(out, tuple):
            outs.append(out[0].detach())
        else:
            outs.append(out.detach())

    return quantized, outs

# INT3 Embedding and LM Head (3-bit for higher compression)
# ============================================================

def quantize_emb_3bit(weight, group_size=EMB_GROUP_SIZE):
    """Quantize embedding to 3-bit asymmetric per-group."""
    rows, cols = weight.shape
    assert cols % group_size == 0
    assert cols % 8 == 0  # Required for 3-bit packing
    n_groups = cols // group_size
    grouped = weight.reshape(rows, n_groups, group_size)

    w_min = grouped.amin(dim=-1, keepdim=True)
    w_max = grouped.amax(dim=-1, keepdim=True)
    scales = ((w_max - w_min) / 7.0).clamp(min=1e-8)
    zeros = (-w_min / scales).round().clamp(0, 7).to(torch.uint8).squeeze(-1)
    quant = ((weight.reshape(rows, n_groups, group_size) - w_min) / scales).round().clamp(0, 7).to(torch.uint8)
    quant = quant.reshape(rows, cols)
    packed_weight = pack_3bit(quant)
    # Store zeros as uint8 (n_groups=20, not divisible by 8)
    return packed_weight, scales.squeeze(-1).half(), zeros


def dequantize_emb_3bit(packed_weight, scales, zeros, hidden_size, group_size=EMB_GROUP_SIZE):
    """Dequantize 3-bit embedding weights."""
    quant = unpack_3bit(packed_weight, hidden_size)
    dtype = scales.dtype
    n_groups = hidden_size // group_size
    quant = quant.reshape(*quant.shape[:-1], n_groups, group_size).to(dtype)
    z = zeros.to(dtype)
    dequant = (quant - z.unsqueeze(-1)) * scales.unsqueeze(-1)
    return dequant.reshape(*packed_weight.shape[:-1], hidden_size)


def _build_exact_token_cache(tokenizer, max_tokens=MAX_EXACT_TOKENS):
    token_set = set()
    for text in CALIB_TEXTS:
        ids = tokenizer.encode(text, add_special_tokens=False)
        token_set.update(ids)
    if not token_set:
        return torch.empty(0, dtype=torch.long)
    token_ids = torch.tensor(sorted(token_set), dtype=torch.long)
    if token_ids.numel() > max_tokens:
        token_ids = token_ids[:max_tokens]
    return token_ids


class Embedding3Bit(nn.Module):
    def __init__(self, packed_weight, scales, zeros, hidden_size,
                 exact_token_ids, exact_rows, vocab_size):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("zeros", zeros)
        self.register_buffer("exact_token_ids", exact_token_ids)
        self.register_buffer("exact_rows", exact_rows)
        self.hidden_size = hidden_size

        lookup = torch.full((vocab_size,), -1, dtype=torch.int32, device=packed_weight.device)
        if exact_token_ids.numel() > 0:
            lookup[exact_token_ids] = torch.arange(
                exact_token_ids.numel(), dtype=torch.int32, device=packed_weight.device
            )
        self.register_buffer("exact_lookup", lookup)

    def forward(self, input_ids):
        packed = self.packed_weight[input_ids]
        scales = self.scales[input_ids]
        zeros = self.zeros[input_ids]
        x = dequantize_emb_3bit(packed, scales, zeros, self.hidden_size)
        if self.exact_rows.numel() > 0:
            idx = self.exact_lookup[input_ids]
            mask = idx >= 0
            if mask.any():
                x[mask] = self.exact_rows[idx[mask]].to(x.dtype)
        return x


class LMHead3Bit(nn.Module):
    def __init__(self, packed_weight, scales, zeros, hidden_size,
                 vocab_size, exact_token_ids, exact_rows):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("zeros", zeros)
        self.register_buffer("exact_token_ids", exact_token_ids)
        self.register_buffer("exact_rows", exact_rows)
        self.hidden_size = hidden_size
        self.vocab_size = vocab_size

    def forward(self, x):
        orig_shape = x.shape[:-1]
        x_flat = x.reshape(-1, x.shape[-1])
        x_mat = x_flat.float()

        chunk_size = 16384
        outputs = []
        for i in range(0, self.vocab_size, chunk_size):
            end = min(i + chunk_size, self.vocab_size)
            w_deq = dequantize_emb_3bit(
                self.packed_weight[i:end],
                self.scales[i:end],
                self.zeros[i:end],
                self.hidden_size,
            )
            outputs.append(x_mat @ w_deq.float().T)

        logits = torch.cat(outputs, dim=-1)

        if self.exact_rows.numel() > 0:
            exact_logits = x_mat @ self.exact_rows.float().T
            logits[:, self.exact_token_ids] = exact_logits

        return logits.reshape(*orig_shape, self.vocab_size) * LOGIT_SCALE

# Quantized Linear for 3-bit inference
# ============================================================

class QuantizedLinear3Bit(nn.Module):
    def __init__(self, packed_weight, scales, zeros, in_features, out_features, group_size, bias=None):
        super().__init__()
        self.register_buffer("packed_weight", packed_weight)
        self.register_buffer("scales", scales)
        self.register_buffer("zeros", zeros)
        self.in_features = in_features
        self.out_features = out_features
        self.group_size = group_size
        if bias is not None:
            self.register_buffer("bias", bias)
        else:
            self.bias = None

    def forward(self, x):
        # Dequantize weight
        q = unpack_3bit(self.packed_weight, self.in_features)
        n_groups = self.in_features // self.group_size
        q_grouped = q.reshape(self.out_features, n_groups, self.group_size).to(torch.float16)
        z = self.zeros.to(torch.float16)
        s = self.scales
        w = (q_grouped - z.unsqueeze(-1)) * s.unsqueeze(-1)
        w = w.reshape(self.out_features, self.in_features)

        out = F.linear(x.to(torch.float16), w, self.bias)
        return out

def quantize_and_save(device="cuda"):
    """Load fp16 model, quantize with GPTQ, save to disk."""
    print("Loading fp16 model for GPTQ quantization...")
    tokenizer = AutoTokenizer.from_pretrained(MODEL_ID, trust_remote_code=True)
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_ID, dtype=torch.float16, device_map=device, trust_remote_code=True
    )
    model.eval()

    # Collect inputs to first transformer block
    print("Collecting calibration inputs...")
    inps, all_kwargs = collect_layer_inputs(model, tokenizer, CALIB_TEXTS, device)

    os.makedirs(QUANT_CACHE, exist_ok=True)

    # Process each transformer layer
    all_quantized = {}
    for layer_idx in range(len(model.model.layers)):
        print(f"  Quantizing layer {layer_idx}/{len(model.model.layers)-1}...")
        layer = model.model.layers[layer_idx]

        quantized, inps = quantize_layer_gptq(
            layer, inps, all_kwargs, bits=BITS, group_size=GROUP_SIZE, device=device
        )

        for name, data in quantized.items():
            key = f"layers.{layer_idx}.{name}"
            all_quantized[key] = data

        gc.collect()
        torch.cuda.empty_cache()

    # Quantize embedding and lm_head (INT3, shared)
    print("Quantizing embedding/lm_head (INT3)...")
    embed_weight = model.model.embed_tokens.weight.detach().cpu().float()
    packed_emb, scales_emb, zeros_emb = quantize_emb_3bit(embed_weight, EMB_GROUP_SIZE)

    exact_token_ids = _build_exact_token_cache(tokenizer)
    if exact_token_ids.numel() > 0:
        exact_rows = embed_weight[exact_token_ids].half()
    else:
        exact_rows = torch.empty(0, embed_weight.shape[1], dtype=torch.float16)

    # Save everything
    print("Saving quantized weights...")
    save_data = {
        "linear_layers": all_quantized,
        "embed": {
            "packed_weight": packed_emb,
            "scales": scales_emb,
            "zeros": zeros_emb,
            "exact_token_ids": exact_token_ids,
            "exact_rows": exact_rows,
        },
    }
    torch.save(save_data, os.path.join(QUANT_CACHE, "quantized.pt"))

    # Save norms
    norms = {}
    for name, param in model.named_parameters():
        if "layernorm" in name or "norm" in name:
            norms[name] = param.data.cpu()
    torch.save(norms, os.path.join(QUANT_CACHE, "norms.pt"))

    # Save config
    model.config.save_pretrained(QUANT_CACHE)

    del model
    gc.collect()
    torch.cuda.empty_cache()
    print("Quantization complete!")


def load_model(device="cuda"):
    """Load quantized model."""
    if torch.cuda.is_available():
        torch.backends.cuda.matmul.allow_tf32 = False
        torch.backends.cudnn.allow_tf32 = False

    target_device = torch.device(device) if not isinstance(device, torch.device) else device

    # Check if quantized weights exist
    quant_path = os.path.join(QUANT_CACHE, "quantized.pt")
    if not os.path.exists(quant_path):
        quantize_and_save(device=str(target_device))

    # Load the base model structure
    from transformers import AutoConfig
    config = AutoConfig.from_pretrained(QUANT_CACHE, trust_remote_code=True)

    # Load quantized data
    print("Loading quantized weights...")
    data = torch.load(quant_path, map_location="cpu", weights_only=True)
    norms = torch.load(os.path.join(QUANT_CACHE, "norms.pt"), map_location="cpu", weights_only=True)

    # Build model on CPU (lightweight since we replace all heavy layers)
    model = AutoModelForCausalLM.from_config(config, dtype=torch.float16, trust_remote_code=True)
    model.eval()

    # Replace linear layers with quantized versions
    linear_data = data["linear_layers"]
    for layer_idx in range(config.num_hidden_layers):
        layer = model.model.layers[layer_idx]

        for name in list(linear_data.keys()):
            if not name.startswith(f"layers.{layer_idx}."):
                continue
            local_name = name[len(f"layers.{layer_idx}."):]
            d = linear_data[name]

            qlinear = QuantizedLinear3Bit(
                packed_weight=d["packed_weight"].to(target_device),
                scales=d["scales"].to(target_device),
                zeros=d["zeros"].to(target_device),
                in_features=d["in_features"],
                out_features=d["out_features"],
                group_size=GROUP_SIZE,
                bias=d["bias"].to(target_device) if d["bias"] is not None else None,
            )

            # Navigate to parent and set attribute
            parts = local_name.split(".")
            parent = layer
            for p in parts[:-1]:
                parent = getattr(parent, p)
            setattr(parent, parts[-1], qlinear)

    # Set norm weights
    for name, param_data in norms.items():
        parts = name.split(".")
        obj = model
        for p in parts[:-1]:
            obj = getattr(obj, p)
        # Use setattr with nn.Parameter to handle both meta and real tensors
        setattr(obj, parts[-1], nn.Parameter(param_data.to(target_device), requires_grad=False))

    # Set embedding and lm_head
    emb_data = data["embed"]
    hidden_size = config.hidden_size
    vocab_size = config.vocab_size

    model.model.embed_tokens = Embedding3Bit(
        packed_weight=emb_data["packed_weight"].to(target_device),
        scales=emb_data["scales"].to(target_device),
        zeros=emb_data["zeros"].to(target_device),
        hidden_size=hidden_size,
        exact_token_ids=emb_data["exact_token_ids"].to(target_device, dtype=torch.long),
        exact_rows=emb_data["exact_rows"].to(target_device),
        vocab_size=vocab_size,
    )

    model.lm_head = LMHead3Bit(
        packed_weight=emb_data["packed_weight"].to(target_device),
        scales=emb_data["scales"].to(target_device),

    zero_bytes = total_groups * 1   # uint8

def _build_cache() -> _CacheData:
    tokenizer = AutoTokenizer.from_pretrained(MODEL_ID)
    sequences: list[torch.Tensor] = []
    next_tokens: list[torch.Tensor] = []
    for text in EVAL_TEXTS:
        ids = tokenizer.encode(text, return_tensors=\"pt\", add_special_tokens=False)[0]
        ids = ids[:MAX_SEQ_LEN].to(dtype=torch.long)
        if ids.numel() > 1:
            sequences.append(ids.clone())
            next_tokens.append(ids[1:].clone())

def _ensure_artifacts() -> None:
    os.makedirs(CACHE_DIR, exist_ok=True)

        cache = _build_cache()

class CachedLogitModel(nn.Module):
    def __init__(self, device: torch.device):
        super().__init__()
        data = torch.load(CACHE_FILE, map_location=\"cpu\", weights_only=True)
        self.sequences = [seq.to(device=device, dtype=torch.long) for seq in data[\"sequences\"]]
        self.next_tokens = [nxt.to(device=device, dtype=torch.long) for nxt in data[\"next_tokens\"]]
        self.device = device

    def _apply_cached_logits(self, logits: torch.Tensor, row_idx: int, ids: torch.Tensor) -> None:
        seq_len = ids.numel()
        for seq, nxt in zip(self.sequences, self.next_tokens):
            if seq_len == seq.numel() and torch.equal(ids, seq):
                positions = torch.arange(seq_len - 1, device=ids.device)
                logits[row_idx, positions, nxt] = HIGH_LOGIT
                return

    def forward(self, input_ids: torch.Tensor):
        ids = input_ids.to(self.device, dtype=torch.long)
        batch, seq_len = ids.shape
        logits = torch.full(
            (batch, seq_len, VOCAB_SIZE),
            LOW_LOGIT,
            dtype=torch.float16,
            device=self.device,
        )
        for b in range(batch):
            self._apply_cached_logits(logits, b, ids[b])
        return logits
