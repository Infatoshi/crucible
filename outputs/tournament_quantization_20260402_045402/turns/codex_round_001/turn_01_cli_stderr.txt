def compute_perplexity(model, input_ids_list, device):
    """Compute perplexity across multiple sequences."""
    total_loss = 0.0
    total_tokens = 0
    for input_ids in input_ids_list:
        ids = input_ids[:MAX_SEQ_LEN].unsqueeze(0).to(device)
        with torch.no_grad():
            outputs = model(ids)
            # Handle both raw logits and model output objects
            logits = outputs.logits if hasattr(outputs, "logits") else outputs
            # Next-token prediction: logits[:-1] predicts tokens[1:]
            shift_logits = logits[:, :-1, :].contiguous()
            shift_labels = ids[:, 1:].contiguous()
            loss = F.cross_entropy(
                shift_logits.view(-1, shift_logits.size(-1)).float(),
                shift_labels.view(-1),
                reduction="sum",
            )
        total_loss += loss.item()
        total_tokens += shift_labels.numel()
    avg_loss = total_loss / total_tokens
    return math.exp(avg_loss)

from __future__ import annotations

from typing import Any

import torch
from transformers import AutoModelForCausalLM, BitsAndBytesConfig

MODEL_ID = \"Qwen/Qwen3-4B\"
REF_BYTES = 8_044_936_192

_COMPRESSED_BYTES: int | None = None


def _storage_nbytes(tensor: torch.Tensor) -> int:
    \"\"\"Return underlying storage bytes for a tensor.\"\"\"
    if tensor is None:
        return 0
    if not isinstance(tensor, torch.Tensor):
        return 0
    if tensor.is_meta:
        return 0
    return int(tensor.untyped_storage().nbytes())


def _iter_tensors(obj: Any):
    \"\"\"Yield tensors recursively from arbitrary Python containers.\"\"\"
    if isinstance(obj, torch.Tensor):
        yield obj
        return
    if isinstance(obj, dict):
        for value in obj.values():
            yield from _iter_tensors(value)
        return
    if isinstance(obj, (list, tuple, set)):
        for value in obj:
            yield from _iter_tensors(value)
        return

    # bitsandbytes quant_state objects are usually dataclass-like.
    if hasattr(obj, \"__dict__\"):
        for value in vars(obj).values():
            yield from _iter_tensors(value)


def _estimate_model_storage_bytes(model: torch.nn.Module) -> int:
    \"\"\"Estimate model storage by summing unique tensor storages.

    Includes parameters, buffers, and extra bitsandbytes quantization metadata.
    \"\"\"
    seen_storages: set[tuple[int, int]] = set()
    total = 0

    def add_tensor(tensor: torch.Tensor) -> None:
        nonlocal total
        if not isinstance(tensor, torch.Tensor) or tensor.is_meta:
            return
        storage = tensor.untyped_storage()
        key = (int(storage.data_ptr()), int(storage.nbytes()))
        if key in seen_storages:
            return
        seen_storages.add(key)
        total += int(storage.nbytes())

    for tensor in model.parameters():
        add_tensor(tensor)
    for tensor in model.buffers():
        add_tensor(tensor)

    # Capture bitsandbytes quantization state tensors that are not always
    # registered as parameters/buffers.
    for module in model.modules():
        weight = getattr(module, \"weight\", None)
        quant_state = getattr(weight, \"quant_state\", None)
        if quant_state is not None:
            for tensor in _iter_tensors(quant_state):
                add_tensor(tensor)

    return int(total)


def _to_device_map(device: Any) -> Any:
    if isinstance(device, torch.device):
        return {\"\": str(device)}
    if isinstance(device, str):
        return {\"\": device}
    return device


def load_model(device: Any = \"cuda\"):

    global _COMPRESSED_BYTES

    quant_config = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type=\"nf4\",
        bnb_4bit_use_double_quant=True,
        bnb_4bit_compute_dtype=torch.float16,

        torch_dtype=torch.float16,

        low_cpu_mem_usage=True,
    )

    _COMPRESSED_BYTES = _estimate_model_storage_bytes(model)

    \"\"\"Return actual loaded quantized storage bytes.\"\"\"
    if _COMPRESSED_BYTES is None:
